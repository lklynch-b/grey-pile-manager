FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build
WORKDIR /src

COPY GreyPileApi.csproj ./
RUN dotnet restore GreyPileApi.csproj

COPY . .
RUN dotnet publish GreyPileApi.csproj -c Release -o /app/publish --no-restore

FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS runtime
WORKDIR /app
COPY --from=build /app/publish .

ENV ASPNETCORE_ENVIRONMENT=Production

# Railway injects $PORT at container start; shell form resolves it at runtime
# (a Dockerfile ENV with ${PORT} would only resolve at build time, before Railway sets it).
ENTRYPOINT dotnet GreyPileApi.dll --urls http://+:$PORT
